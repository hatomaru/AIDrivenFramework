using System;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;

public static class SetupValidation
{
    [InitializeOnLoadMethod]
    static void AfterImportReload()
    {
        if (SessionState.GetBool("SetupValidation.ImportPending", false))
            EditorApplication.delayCall += FinishImport;
    }
    static void FinishImport()
    { SessionState.SetBool("SetupValidation.ImportPending", false); EditorApplication.Exit(0); }
    public static void Import()
    {
        string package = Environment.GetEnvironmentVariable("SETUP_PACKAGE_PATH");
        if (string.IsNullOrEmpty(package)) throw new ArgumentException("SETUP_PACKAGE_PATH is required.");
        SessionState.SetBool("SetupValidation.ImportPending", true);
        AssetDatabase.importPackageCompleted += _ => EditorApplication.delayCall += FinishImport;
        AssetDatabase.ImportPackage(package, false);
        AssetDatabase.Refresh();
    }
    public static void Build()
    {
        PlayerSettings.companyName = "FrameworkValidation";
        PlayerSettings.productName = "PlayerSetupValidation";
        PlayerSettings.defaultScreenWidth = 1440;
        PlayerSettings.defaultScreenHeight = 900;
        PlayerSettings.fullScreenMode = FullScreenMode.Windowed;
        PlayerSettings.runInBackground = true;
        PlayerSettings.SetScriptingBackend(UnityEditor.Build.NamedBuildTarget.Standalone, ScriptingImplementation.Mono2x);
        var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        new GameObject("Smoke Runner").AddComponent(Type.GetType("SetupSmokeRunner, Assembly-CSharp"));
        var camera = new GameObject("Camera").AddComponent<Camera>();
        camera.clearFlags = CameraClearFlags.SolidColor;
        camera.backgroundColor = Color.white;
        EditorSceneManager.SaveScene(scene, "Assets/TestGame.unity");
        string[] scenes = { "Assets/TestGame.unity", "Assets/AIDrivenFW/AISetup/AIDrivenSetup.unity" };
        EditorBuildSettings.scenes = new[] { new EditorBuildSettingsScene(scenes[0], true), new EditorBuildSettingsScene(scenes[1], true) };
        var result = BuildPipeline.BuildPlayer(new BuildPlayerOptions
        { scenes = scenes, target = BuildTarget.StandaloneWindows64, locationPathName = "Build/SetupValidation.exe", options = BuildOptions.Development });
        if (result.summary.result != UnityEditor.Build.Reporting.BuildResult.Succeeded)
            throw new Exception("Validation player build failed.");
    }
}
