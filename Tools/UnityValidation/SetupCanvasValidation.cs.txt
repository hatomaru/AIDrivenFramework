        public void RunFinished(ITestResultAdaptor result)
        {
            TestRunnerApi.SaveResultToFile(result, Path.Combine(Output, "unit.xml"));
            Debug.Log("SETUP UNIT RESULTS: " + result.PassCount + " passed; " + result.FailCount + " failed; " + result.SkipCount + " skipped");
            if (SessionState.GetBool("CanvasValidation.AfterUnits", false))
            {
                SessionState.SetBool("CanvasValidation.AfterUnits", false);
                if (result.FailCount == 0 && result.PassCount > 0) EditorApplication.delayCall += Start;
            }
        }

    {
        if (EditorApplication.isPlaying) throw new InvalidOperationException("Stop Play before validating.");
        if (!UnityEngine.Object.FindFirstObjectByType<PlayerSetupScreen>())

            if (UnityEngine.SceneManagement.SceneManager.GetActiveScene().isDirty)
                throw new InvalidOperationException("Save the current scene before validating.");
            UnityEditor.SceneManagement.EditorSceneManager.OpenScene("Assets/AIDrivenFW/AISetup/AIDrivenSetup.unity");
        }
        SessionState.SetBool("CanvasValidation.Pending", true); EditorApplication.isPlaying = true;

    static double Contrast(Color a, Color b)
    {
        double Linear(float c) => c <= .04045 ? c / 12.92 : Math.Pow((c + .055) / 1.055, 2.4);
        double Lum(Color c) => Linear(c.r) * .2126 + Linear(c.g) * .7152 + Linear(c.b) * .0722;
        return (Math.Max(Lum(a), Lum(b)) + .05) / (Math.Min(Lum(a), Lum(b)) + .05);
    }
