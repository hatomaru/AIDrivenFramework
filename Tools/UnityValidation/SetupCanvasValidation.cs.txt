using System;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Threading;
using AIDrivenFW.Config;
using AIDrivenFW.Setup;
using Cysharp.Threading.Tasks;
using UnityEditor;
using UnityEngine;
using UnityEngine.UI;
using UnityEngine.EventSystems;
using System.Collections.Generic;

// Install in Assets/Editor of the validation host. Uses inert files and injected services only.
public static class SetupCanvasValidation
{
    const BindingFlags Private = BindingFlags.Instance | BindingFlags.NonPublic;
    static int assertions;
    static readonly string Output = Environment.GetEnvironmentVariable("SETUP_QA_OUTPUT") ?? Path.Combine(Path.GetTempPath(), "FrameworkCanvasProof");
    [MenuItem("Tools/AIDrivenFW/Canvas Setup/Validate Offline")]
    public static void Start()
    {
        if (EditorApplication.isPlaying) throw new InvalidOperationException("Stop Play before validating.");
        SessionState.SetBool("CanvasValidation.Pending", true); EditorApplication.isPlaying = true;
    }
    [InitializeOnLoadMethod]
    static void Register()
    {
        EditorApplication.playModeStateChanged += state =>
        {
            if (state == PlayModeStateChange.EnteredPlayMode && SessionState.GetBool("CanvasValidation.Pending", false))
            { SessionState.SetBool("CanvasValidation.Pending", false); Run().Forget(); }
        };
    }
    static void Check(bool ok, string name) { if (!ok) throw new Exception(name); assertions++; Debug.Log("CANVAS CHECK: " + name); }
    static async UniTask Run()
    {
        Directory.CreateDirectory(Output); bool passed = false; string path = Environment.GetEnvironmentVariable("PATH");
        string language = PlayerPrefs.GetString("AIDrivenFW.Setup.Language", "en"); int motion = PlayerPrefs.GetInt("AIDrivenFW.Setup.ReduceMotion", 0);
        try
        {
            await UniTask.Delay(500, ignoreTimeScale: true); assertions = 0;
            var ui = UnityEngine.Object.FindFirstObjectByType<PlayerSetupScreen>(); Check(ui && ui.enabled, "Authored scene loaded");
            Check(ui.Slides.Length == 8 && ui.Browser.Rows.Length == 8, "Eight authored slides and reusable browser rows");
            foreach (var field in typeof(PlayerSetupScreen).GetFields(BindingFlags.Public | BindingFlags.Instance))
                if (typeof(UnityEngine.Object).IsAssignableFrom(field.FieldType)) Check(field.GetValue(ui) is UnityEngine.Object o && o, "Assigned: " + field.Name);
            Check(ui.GetComponentsInChildren<Text>(true).All(t => t.font && t.font.HasCharacter('語') && t.font.HasCharacter('A')), "All labels support Japanese and English");
            var services = new Fixture(); ui.SetServices(services); ui.SetLanguage(true); ui.Go(PlayerSetupScreen.Slide.Welcome);
            Check(ui.PrimaryCaption.text == "続ける", "Japanese navigation"); ui.SetLanguage(false); Check(ui.PrimaryCaption.text == "Continue", "English navigation"); ui.SetLanguage(true);
            ui.ReduceMotionToggle.isOn = false; Check(!ui.ReduceMotionToggle.isOn, "Motion enabled");
            Check(EventSystem.current && EventSystem.current.currentInputModule, "Active UI input module exists");
            Canvas.ForceUpdateCanvases(); var pointer = new PointerEventData(EventSystem.current);
            pointer.position = RectTransformUtility.WorldToScreenPoint(null, ui.ChangeModelButton.GetComponent<RectTransform>().TransformPoint(ui.ChangeModelButton.GetComponent<RectTransform>().rect.center));
            var hits = new List<RaycastResult>(); EventSystem.current.RaycastAll(pointer, hits);
            Check(hits.Count > 0 && hits[0].gameObject.GetComponentInParent<Button>() == ui.ChangeModelButton, "Authored navigation receives pointer raycasts");
            ExecuteEvents.Execute(ui.ChangeModelButton.gameObject, pointer, ExecuteEvents.pointerClickHandler);
            Check(ui.CurrentSlide == PlayerSetupScreen.Slide.Runtime, "Pointer handler opens runtime selection"); ui.Go(PlayerSetupScreen.Slide.Welcome);
            var fixture = Path.Combine(Output, "Files"); Directory.CreateDirectory(fixture); Directory.CreateDirectory(Path.Combine(fixture, "Models"));
            File.WriteAllText(Path.Combine(fixture, "llama-server.exe"), "inert fixture; never execute"); File.WriteAllText(Path.Combine(fixture, "fixture.gguf"), "inert fixture; never load"); File.WriteAllText(Path.Combine(fixture, "other.exe"), "inert fixture");
            Environment.SetEnvironmentVariable("PATH", fixture + Path.PathSeparator + path);
            typeof(PlayerSetupScreen).GetField("installedModels", Private).SetValue(ui, new[] { new InstalledModel { name = "qwen3:0.6b", size = 523000000 } });
            ui.PrimaryButton.onClick.Invoke(); Check(ui.CurrentSlide == PlayerSetupScreen.Slide.Runtime, "Both runtimes require an explicit choice");
            ui.LlamaButton.onClick.Invoke(); Check(ui.CurrentSlide == PlayerSetupScreen.Slide.Custom && ui.LlamaFields.activeSelf, "llama.cpp uses authored file form");
            ui.RuntimePath.text = Path.Combine(fixture, "llama-server.exe"); ui.ModelPath.text = Path.Combine(fixture, "fixture.gguf");
            ui.BrowseRuntimeButton.onClick.Invoke(); var browser = ui.Browser; await WaitBrowser(browser);
            Check(browser.Rows.Any(r => r.Button.gameObject.activeSelf && r.Name.text == "fixture.gguf" && !r.Button.interactable), "GGUF cannot be chosen as runtime");
            Check(!browser.Rows.Any(r => r.Button.gameObject.activeSelf && r.Name.text == "other.exe"), "Unrelated executable excluded");
            Check(browser.Rows.Where(r => r.Button.gameObject.activeSelf).All(r => r.Icon.sprite), "Recognized file types have authored icons");
            ui.SecondaryButton.onClick.Invoke(); ui.BrowseModelButton.onClick.Invoke(); await WaitBrowser(browser);
            Check(browser.Rows.Any(r => r.Button.gameObject.activeSelf && r.Name.text == "llama-server.exe" && !r.Button.interactable), "Runtime cannot be chosen as GGUF");
            browser.ReaderOverride = (p, f, n, ct, progress) => { progress(64); Thread.Sleep(450); return new[] { new SetupFilePicker.Entry { Name = "late.gguf", Path = "late.gguf", Kind = SetupFileKind.Model } }; };
            browser.OpenButton.onClick.Invoke(); await UniTask.Delay(80); Check(browser.IsLoading && browser.LoadingTrack.activeSelf, "Slow directory shows loading bar and stays interactive");
            ui.SecondaryButton.onClick.Invoke(); await UniTask.Delay(500); Check(!browser.IsLoading && ui.CurrentSlide == PlayerSetupScreen.Slide.Custom, "Back cancels and stale results stay closed");
            browser.TimeoutSeconds = .1f; ui.BrowseModelButton.onClick.Invoke(); await UniTask.Delay(250); Check(!browser.IsLoading && browser.Status.text.Contains("時間"), "Slow read times out with recovery message");
            browser.ReaderOverride = null; browser.TimeoutSeconds = 15; browser.OpenButton.onClick.Invoke(); await WaitBrowser(browser); Check(browser.Rows.Any(r => r.Button.gameObject.activeSelf), "Directory can be retried after timeout");
            ui.SecondaryButton.onClick.Invoke(); ui.PrimaryButton.onClick.Invoke(); await UniTask.Delay(300);
            Check(ui.CurrentSlide == PlayerSetupScreen.Slide.Consent && !ui.ConsentToggle.isOn && !ui.PrimaryButton.interactable, "Local files reach explicit consent without URL entry");
            Check(ui.PrimaryButton.GetComponent<Image>().sprite == ui.ButtonShape, "Disabled consent button uses a neutral surface");
            Check(Contrast(ui.PrimaryCaption.color, ui.PrimaryButton.colors.disabledColor) >= 4.5, "Disabled consent caption contrast at least 4.5:1");
            Check(!ui.ModelLicenseButton.gameObject.activeSelf && ui.ConsentCaption.text.Contains("入手元"), "Unknown GGUF requires source review without fabricated terms");
            Check(services.Calls == 0 && services.Settings == null, "No preparation or commit before consent");
            ui.ConsentToggle.isOn = true; ui.PrimaryButton.onClick.Invoke(); await UniTask.Delay(180); Check(ui.CurrentSlide == PlayerSetupScreen.Slide.Progress, "Consent starts progress slide");
            ui.SecondaryButton.onClick.Invoke(); await UniTask.Delay(180); Check(ui.Flow.Stage == SetupStage.Cancelled && services.Settings == null, "Pause prevents saving settings");
            ui.ChangeModelButton.onClick.Invoke(); ui.OllamaButton.onClick.Invoke(); Check(ui.CurrentSlide == PlayerSetupScreen.Slide.Models, "Ollama opens configured model choices");
            var choices = PlayerSetupPlan.Choices(); Check(ui.ModelSlots[0].Name.text == choices[0].Model, "Model information comes from config");
            ui.ModelSlots[0].Button.onClick.Invoke(); ui.PrimaryButton.onClick.Invoke(); Check(ui.CurrentSlide == PlayerSetupScreen.Slide.Consent, "Ollama selection reaches review");
            services = new Fixture(); ui.SetServices(services); ui.ConsentToggle.isOn = true; ui.PrimaryButton.onClick.Invoke(); await UniTask.Delay(1350);
            Check(ui.DownloadedBytes.text.Contains("251") && ui.RemainingBytes.text.Contains("272"), "Download shows 251 MB received and 272 MB remaining");
            await UniTask.Delay(1700); Check(services.Settings != null && services.Verified && ui.Flow.Stage == SetupStage.Complete, "Only verified selection commits");
            Check(ui.CurrentSlide == PlayerSetupScreen.Slide.Result && ui.ResultFill.fillAmount == 1, "Completion uses authored result slide");
            File.WriteAllText(Path.Combine(Output, "result.txt"), "PASS: " + assertions + " Canvas assertions\nNo model acquisition or native inference was performed.\n"); Debug.Log("CANVAS VALIDATION PASS: " + assertions); passed = true;
        }
        catch (Exception e) { Debug.LogException(e); File.WriteAllText(Path.Combine(Output, "result.txt"), "FAIL: " + e); }
        finally
        {
            Environment.SetEnvironmentVariable("PATH", path); PlayerPrefs.SetString("AIDrivenFW.Setup.Language", language); PlayerPrefs.SetInt("AIDrivenFW.Setup.ReduceMotion", motion); PlayerPrefs.Save(); EditorApplication.isPlaying = false;
            if (Application.isBatchMode) EditorApplication.delayCall += () => EditorApplication.Exit(passed ? 0 : 1);
        }
    }
    static async UniTask WaitBrowser(SetupFilePicker b) { for (int i = 0; b.IsLoading && i < 100; i++) await UniTask.Delay(30); Check(!b.IsLoading, "Directory read finished"); }
    static double Contrast(Color a, Color b)
    {
        double Linear(float c) => c <= .04045 ? c / 12.92 : Math.Pow((c + .055) / 1.055, 2.4);
        double Lum(Color c) => Linear(c.r) * .2126 + Linear(c.g) * .7152 + Linear(c.b) * .0722;
        return (Math.Max(Lum(a), Lum(b)) + .05) / (Math.Min(Lum(a), Lum(b)) + .05);
    }
    sealed class Fixture : IPlayerSetupServices
    {
        public int Calls; public bool Verified; public ModelInfo Settings;
        public async UniTask<string> PrepareRuntimeAsync(PlayerSetupPlan p, Action<SetupProgress> report, CancellationToken ct) { Calls++; report(new SetupProgress { Stage = SetupStage.Runtime, CompletedBytes = 50000000, TotalBytes = 100000000 }); await UniTask.Delay(1000, cancellationToken: ct); return p.RuntimePath; }
        public async UniTask PrepareModelAsync(PlayerSetupPlan p, Action<SetupProgress> report, CancellationToken ct) { Calls++; report(new SetupProgress { Stage = SetupStage.Model, CompletedBytes = 251000000, TotalBytes = 523000000 }); await UniTask.Delay(1500, cancellationToken: ct); }
        public async UniTask VerifyAsync(PlayerSetupPlan p, CancellationToken ct) { await UniTask.Delay(200, cancellationToken: ct); Verified = true; }
        public void RecordAttempt(PlayerSetupPlan p, string utc, SetupStage stage) { }
        public void Commit(ModelInfo settings) { if (!Verified) throw new InvalidOperationException("Verification required"); Settings = settings; }
        public void Dispose() { }
    }
}
