using System;
using System.IO;
using System.Linq;
using AIDrivenFW.API;
using AIDrivenFW.Config;
using AIDrivenFW.Setup;
using Cysharp.Threading.Tasks;
using UnityEngine;
using UnityEngine.SceneManagement;

// Live validation only: invoked explicitly by run.ps1 without -DisplayOnly.
public sealed class SetupSmokeRunner : MonoBehaviour
{
    bool finished, result;
    string output;
    async UniTaskVoid Start()
    {
        output = Environment.GetEnvironmentVariable("SETUP_QA_OUTPUT") ?? Path.Combine(Application.persistentDataPath, "ValidationProof");
        Directory.CreateDirectory(output);
        string settings = Read(ModelInfo.SettingsPath), journal = Read(SetupJournal.FilePath);
        string language = PlayerPrefs.GetString("AIDrivenFW.Setup.Language", "en"); int motion = PlayerPrefs.GetInt("AIDrivenFW.Setup.ReduceMotion", 0);
        int exit = 1;
        try
        {
            Restore(ModelInfo.SettingsPath, null); Restore(SetupJournal.FilePath, null);
            PlayerPrefs.SetString("AIDrivenFW.Setup.Language", "en"); PlayerPrefs.SetInt("AIDrivenFW.Setup.ReduceMotion", 0);
            Initialize(); await Wait(() => SceneManager.GetSceneByName("AIDrivenSetup").isLoaded);
            var ui = Find(); Check(ui.Slides.Count(s => s.activeSelf) == 1, "one slide");
            Check(!ui.ConsentToggle.isOn && !ui.ReduceMotionToggle.isOn, "consent off and motion on");
            await Capture("canvas-welcome-en"); ui.JapaneseButton.onClick.Invoke(); await Capture("canvas-welcome-ja");
            ui.ChangeModelButton.onClick.Invoke(); ui.OllamaButton.onClick.Invoke(); ui.ModelSlots[0].Button.onClick.Invoke(); ui.PrimaryButton.onClick.Invoke();
            Check(ui.CurrentSlide == PlayerSetupScreen.Slide.Consent && !ui.PrimaryButton.interactable, "consent gate"); await Capture("canvas-consent-ja");
            ui.SecondaryButton.onClick.Invoke(); ui.SecondaryButton.onClick.Invoke(); await Wait(() => finished);
            Check(!result && !File.Exists(ModelInfo.SettingsPath), "refusal has no ready settings");
            Initialize(); await Wait(() => SceneManager.GetSceneByName("AIDrivenSetup").isLoaded); ui = Find();
            ui.ChangeModelButton.onClick.Invoke(); ui.OllamaButton.onClick.Invoke(); ui.ModelSlots[0].Button.onClick.Invoke(); ui.PrimaryButton.onClick.Invoke();
            ui.ConsentToggle.isOn = true; ui.PrimaryButton.onClick.Invoke();
            await Wait(() => finished, 240); Check(result, "real generation and additive return");
            Check(ModelInfo.LoadFromFile()?.HasPlayerConsent == true, "verified consent persisted");
            Initialize(); await Wait(() => finished, 240); Check(result && !SceneManager.GetSceneByName("AIDrivenSetup").isLoaded, "ready setup skips scene");
            File.WriteAllText(Path.Combine(output, "result.txt"), "PASS: live Ollama generation, explicit consent, refusal, additive return and ready skip.\n"); exit = 0;
        }
        catch (Exception e) { File.WriteAllText(Path.Combine(output, "result.txt"), "FAIL: " + e); Debug.LogException(e); }
        finally
        {
            Restore(ModelInfo.SettingsPath, settings); Restore(SetupJournal.FilePath, journal);
            PlayerPrefs.SetString("AIDrivenFW.Setup.Language", language); PlayerPrefs.SetInt("AIDrivenFW.Setup.ReduceMotion", motion); PlayerPrefs.Save();
#if UNITY_EDITOR
            UnityEditor.EditorApplication.Exit(exit);
#else
            Application.Quit(exit);
#endif
        }
    }
    void Initialize() { finished = false; AIDrivenInitializer.Initialize().ContinueWith(value => { result = value; finished = true; }).Forget(); }
    static PlayerSetupScreen Find() => SceneManager.GetSceneByName("AIDrivenSetup").GetRootGameObjects().Select(g => g.GetComponent<PlayerSetupScreen>()).First(s => s);
    static void Check(bool ok, string message) { if (!ok) throw new Exception(message); Debug.Log("LIVE CANVAS PASS: " + message); }
    static async UniTask Wait(Func<bool> ready, int seconds = 30) { float start = Time.realtimeSinceStartup; while (!ready()) { if (Time.realtimeSinceStartup - start > seconds) throw new TimeoutException(); await UniTask.Yield(); } }
    async UniTask Capture(string name) { await UniTask.Delay(600); ScreenCapture.CaptureScreenshot(Path.Combine(output, name + ".png")); await UniTask.Delay(200); }
    static string Read(string path) => File.Exists(path) ? File.ReadAllText(path) : null;
    static void Restore(string path, string text) { if (text != null) ModelInfo.WriteAtomic(path, text); else if (File.Exists(path)) File.Delete(path); }
}
